cmake_minimum_required(VERSION 3.19)
project(qwk_component_consumer LANGUAGES CXX)

list(APPEND CMAKE_PREFIX_PATH "${QT_PREFIX}")
set(package_path CONFIG PATHS "${PACKAGE_DIR}" NO_DEFAULT_PATH)
set(expected_components Core)
if(USE_WIDGETS)
    list(APPEND expected_components Widgets)
endif()
if(USE_QUICK)
    list(APPEND expected_components Quick)
endif()
set(missing MissingComponent)
if(NOT USE_WIDGETS)
    set(missing Widgets)
elseif(NOT USE_QUICK)
    set(missing Quick)
endif()
set(link_components Core)
if(CASE STREQUAL "present")
    find_package(QWindowKit REQUIRED COMPONENTS ${expected_components} ${package_path})
    set(link_components "${expected_components}")
elseif(CASE STREQUAL "default")
    find_package(QWindowKit REQUIRED ${package_path})
    set(link_components "${expected_components}")
elseif(CASE STREQUAL "core")
    find_package(QWindowKit REQUIRED COMPONENTS Core ${package_path})
elseif(CASE MATCHES "^(missing-required|unknown-required|quiet-missing|repeat)$")
    if(CASE STREQUAL "unknown-required")
        set(missing MisspelledComponent)
    endif()
    # A missing QWK component must be reported before any Qt dependency lookup.
    set(CMAKE_DISABLE_FIND_PACKAGE_Qt${QT_MAJOR} TRUE)
    if(CASE MATCHES "required$")
        find_package(QWindowKit REQUIRED COMPONENTS ${missing} ${package_path})
        message(FATAL_ERROR "Missing required component was accepted")
    endif()
    find_package(QWindowKit QUIET COMPONENTS ${missing} ${package_path})
    if(QWindowKit_FOUND OR QWindowKit_${missing}_FOUND OR TARGET QWindowKit::Core OR
       NOT QWindowKit_NOT_FOUND_MESSAGE MATCHES "Unavailable QWindowKit components:")
        message(FATAL_ERROR "Incorrect missing-component state/diagnostic")
    endif()
    if(CASE STREQUAL "quiet-missing")
        return()
    endif()
    unset(CMAKE_DISABLE_FIND_PACKAGE_Qt${QT_MAJOR})
    find_package(QWindowKit REQUIRED COMPONENTS Core ${package_path})
    if(QWindowKit_NOT_FOUND_MESSAGE)
        message(FATAL_ERROR "A preceding failed lookup leaked its diagnostic")
    endif()
elseif(CASE STREQUAL "optional")
    find_package(QWindowKit REQUIRED COMPONENTS Core
        OPTIONAL_COMPONENTS ${missing} AnotherUnknown ${package_path})
    if(QWindowKit_${missing}_FOUND OR QWindowKit_AnotherUnknown_FOUND)
        message(FATAL_ERROR "Absent optional components were reported as found")
    endif()
elseif(CASE STREQUAL "optional-qt-missing")
    if(USE_WIDGETS)
        set(optional Widgets)
    elseif(USE_QUICK)
        set(optional Quick)
    else()
        set(optional Widgets)
    endif()
    set(CMAKE_DISABLE_FIND_PACKAGE_Qt${QT_MAJOR}${optional} TRUE)
    find_package(QWindowKit REQUIRED COMPONENTS Core OPTIONAL_COMPONENTS ${optional} ${package_path})
    if(QWindowKit_${optional}_FOUND)
        message(FATAL_ERROR "Optional component without its Qt dependency was reported as usable")
    endif()
else()
    message(FATAL_ERROR "Unknown test case: ${CASE}")
endif()

if(NOT QWindowKit_FOUND OR NOT QWindowKit_Core_FOUND)
    message(FATAL_ERROR "Required Core/package not found")
endif()
if(NOT QWindowKit_AVAILABLE_COMPONENTS STREQUAL expected_components)
    message(FATAL_ERROR "Installed manifest differs: ${QWindowKit_AVAILABLE_COMPONENTS} vs ${expected_components}")
endif()
if(NOT QWindowKit_QT_VERSION_MAJOR EQUAL QT_MAJOR)
    message(FATAL_ERROR "Installed package lost its build-time Qt major")
endif()
foreach(component Core Widgets Quick)
    if(component IN_LIST expected_components)
        if(NOT TARGET QWindowKit::${component})
            message(FATAL_ERROR "Installed target ${component} is absent")
        endif()
    elseif(TARGET QWindowKit::${component} OR QWindowKit_${component}_FOUND)
        message(FATAL_ERROR "Uninstalled component ${component} leaked into this consumer")
    endif()
endforeach()
add_executable(component_consumer ../../consumer.cpp)
target_compile_features(component_consumer PRIVATE cxx_std_17)
foreach(component IN LISTS link_components)
    if(NOT QWindowKit_${component}_FOUND)
        message(FATAL_ERROR "Requested ${component} is not usable")
    endif()
    target_link_libraries(component_consumer PRIVATE QWindowKit::${component})
    string(TOUPPER "${component}" upper)
    target_compile_definitions(component_consumer PRIVATE CONSUMER_USE_${upper})
endforeach()
