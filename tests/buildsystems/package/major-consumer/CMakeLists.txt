cmake_minimum_required(VERSION 3.19)
project(qwk_major_contract LANGUAGES NONE)

# Search only the explicitly supplied fake package roots, never a local Qt installation.
set(CMAKE_FIND_USE_CMAKE_ENVIRONMENT_PATH FALSE)
set(CMAKE_FIND_USE_SYSTEM_ENVIRONMENT_PATH FALSE)
set(CMAKE_FIND_USE_CMAKE_SYSTEM_PATH FALSE)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY FALSE)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY FALSE)
set(CMAKE_PREFIX_PATH "${QT_ROOTS}")
set(QT_VERSION_MAJOR "${WRONG_MAJOR}")
find_package(QWindowKit REQUIRED COMPONENTS Core Widgets Quick)
get_property(found_majors GLOBAL PROPERTY QWK_TEST_QT_MAJORS)
foreach(major IN LISTS found_majors)
    if(NOT major EQUAL EXPECTED_MAJOR)
        message(FATAL_ERROR "Selected Qt ${major}, expected Qt ${EXPECTED_MAJOR}")
    endif()
endforeach()
if(NOT QWindowKit_QT_VERSION_MAJOR EQUAL EXPECTED_MAJOR)
    message(FATAL_ERROR "The package did not report its build-time Qt major")
endif()
foreach(module Core Gui Widgets Quick)
    if(NOT TARGET Qt${EXPECTED_MAJOR}::${module} OR TARGET Qt${WRONG_MAJOR}::${module})
        message(FATAL_ERROR "Wrong public Qt targets for ${module}")
    endif()
    if(STATIC_PACKAGE AND EXPECTED_MAJOR EQUAL 6 AND NOT TARGET Qt6::${module}Private)
        message(FATAL_ERROR "Missing static Qt dependency: ${module}Private")
    endif()
endforeach()
foreach(component Core Widgets Quick)
    if(NOT TARGET QWindowKit::${component})
        message(FATAL_ERROR "Missing exported ${component} target")
    endif()
endforeach()
